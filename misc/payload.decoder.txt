//  change voltage resolution to from 12-bit to 8-bit
// 20251124/pbiering: add support for "extra info v2"
//  add txpower (4-bit) - unchanged to v1
//  add datarate (4-bit) - unchanged to v1
//  add changed (1-bit) - unchanged to v1
//  add period (4-bit) with resolution of 3 - changed to v1
//  add threshold (8-bit) to resolution of 16 - changed to v1
//  revert voltage resolution from 8-bit to 12-bit - changed to v1

function decodeUplink(input) {
  // Decode an uplink message from a buffer
  // (array) of bytes to an object of fields.
  var decoded = {};

  if ((input.bytes[0] & 0x80) === 0x80) decoded.box = "full";
  else decoded.box = "empty";
  decoded.voltage = ((input.bytes[2]<<8) + input.bytes[1])/1000;
  decoded.sensor1 = (input.bytes[4]<<8) + input.bytes[3];
  decoded.sensor2 = (input.bytes[6]<<8) + input.bytes[5];
  decoded.threshold = input.bytes[7];

  // extra info
  if ((input.bytes[0] & 0x30) === 0x10) {
    decoded.version = input.bytes[0] & 0x0F;
    if ((input.bytes[0] & 0x0F) === 1) {
      // extra info encoded version 1
      decoded.datarate = (decoded.sensor1 & 0xF000)>>>12;
      decoded.sensor1  &= 0x0FFF;
      decoded.txpower  = (decoded.sensor2 & 0xF000)>>>12;
      decoded.sensor2  &= 0x0FFF;
      decoded.changed  = (input.bytes[0] & 0x40)>>>6;
      decoded.voltage  = (1500 + input.bytes[1]*10)/1000;
      decoded.period   = input.bytes[2];
    }
    if ((input.bytes[0] & 0x0F) === 2) {
      // extra info encoded version 2
      decoded.datarate = (decoded.sensor1 & 0xF000)>>>12;
      decoded.sensor1  &= 0x0FFF;
      decoded.txpower  = (decoded.sensor2 & 0xF000)>>>12;
      decoded.sensor2  &= 0x0FFF;
      decoded.changed  = (input.bytes[0] & 0x40)>>>6;
      decoded.voltage  = (input.bytes[1] + (input.bytes[2] & 0x0F) * 256) / 1000;
      decoded.period   = ((input.bytes[2] & 0xF0)>>>4) * 3;
      decoded.threshold = input.bytes[7] * 16;
    }
  }

  return {
    data: decoded,
    warnings: [],
    errors: []
  };
}
